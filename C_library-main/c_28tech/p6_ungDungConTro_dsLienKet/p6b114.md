#include <stdio.h>
#include <stdlib.h>

struct node{
    int data;
    struct node *next;//link
};

typedef struct node node;

node* makeNode(int x){
    //Cap phat dong 1 node
    node *newNode = (node*)malloc(sizeof(node));
    newNode->data = x;
    newNode->next = NULL;
    return newNode;
}

void pushFront(node **head, int x){
    node *newNode = makeNode(x);
    newNode->next = (*head); // toan tu giai tham chieu
    (*head) = newNode;
}

void duyet(node *head){
    while(head != NULL){
        printf("%d ", head->data);
        //dich head sang node ben phai
        head = head->next;//quan trong
    }
    printf("\n");
}

int size(node *head){
    int cnt = 0;
    while(head != NULL){
        ++cnt;
        head = head->next;
    }
    return cnt;
}

int main(){
    node *head = NULL;
    pushFront(&head, 1);
    pushFront(&head, 2);
    pushFront(&head, 3);
    pushFront(&head, 4);
    duyet(head);
    return 0;
}
