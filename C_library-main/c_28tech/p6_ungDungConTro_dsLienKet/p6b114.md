		head = head->next;//quan trong

}

int size(node *head) {
	int cnt = 0;
	while(head != NULL) {
		++cnt;
		head = head->next;
	}
	return cnt;
}

node* makeNode(int x) {
	//Cap phat dong 1 node
	node *newNode = (node*)malloc(sizeof(node));
	newNode->data = x;
	newNode->next = NULL;
	return newNode;
}

void pushFront(node **head, int x) {
	node *newNode = makeNode(x);
	newNode->next = (*head); // toan tu giai tham chieu
	(*head) = newNode;
}

void pushBack(node **head, int x) {
	node *newNode = makeNode(x);
	node *temp = (*head);
	if(temp == NULL){
	    (*head) = newNode; return;
	}
	while(temp->next != NULL) {
		temp = temp->next;
	}
	temp->next = newNode;
}

// chen vao vi tri k trong danh sach lien ket
void insert(node ** head, int k, int x){
    int n = size(*head);
    if(k < 1 || k > n + 1){
        printf("Vi tri chen khong hop le!\n");
        return;
    }
    if(k == 1){
        pushFront(head, x);
    }
    else if(k == n + 1){
        pushBack(head, x);
    }
    else{
        node *temp = (*head);
        for(int i = 1; i < k - 1; i++){
            temp = temp->next;
        }
        node *newNode = makeNode(x);
        //Buoc 1: phan next cua new Node luu dia chi thu k
        newNode->next = temp->next;
        //Buoc 2: phan next cua node thu k - 1 luu dia chi cua node moi
        temp->next = newNode;
    }
}

int main() {
	node *head = NULL;
	while(1) {
		printf("---------------------------------\n");
		printf("1. Them vao cuoi\n");
		printf("2. Them vao dau\n");
		printf("3. Chen vao giua\n");
		printf("4. Duyet DSLK\n");

            int k; scanf("%d", &k);

            duyet(head);
        }
